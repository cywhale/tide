## Appendix: `09b_py311_suite.sh`

sha256 `42249079d9030cef313195267833c0200992952eb12359dae0b9f2c1b12d1c65`,
the same file on VM24 and on the operator's machine.

```text
# Step 9, second part: the py311 suite of the approved commit as in release
# sequence step 2 -- a separate checkout (never $SRC), the two stores linked
# read-only as symlinks, the VM24 py311 interpreter, nothing installed.
set -euo pipefail
M=64361922fd169a535cd1d9759459ba21660f244f
B=/home/odbadmin/tmp/tide-zarr-6436192.bundle
P=/home/odbadmin/tmp/tide-zarr-6436192-pytest
PY=/home/odbadmin/.pyenv/versions/py311/bin/python
[ ! -e "$P" ] || { echo "FAIL suite: $P already exists" >&2; exit 1; }
git init -q "$P"
git -C "$P" bundle verify "$B" > /dev/null
git -C "$P" fetch -q "$B" refs/remotes/origin/main:refs/heads/main
git -C "$P" checkout -q --detach "$M"
[ "$(git -C "$P" rev-parse HEAD)" = "$M" ] || { echo "FAIL suite: $P is not at $M" >&2; exit 1; }
[ -z "$(git -C "$P" status --porcelain)" ] || { echo "FAIL suite: $P is not clean" >&2; exit 1; }
mkdir "$P/data"
ln -s /home/odbadmin/python/tide_tpxo10_atlas_v2/data/tpxo10.zarr "$P/data/tpxo10.zarr"
ln -s /home/odbadmin/python/tide/data/tpxo9.zarr "$P/data/tpxo9.zarr"
ls -l "$P/data"
cd "$P"
echo "zarr in py311: $("$PY" -I -c 'import zarr; print(zarr.__version__)')"
rc=0
TIDE_DASK_DISABLE=1 "$PY" -m pytest -rs || rc=$?
echo "suite exit $rc"
prc=0
TIDE_DASK_DISABLE=1 "$PY" -m pytest -v tests/test_app_smoke.py::test_pinned_runtime_versions || prc=$?
echo "pin test exit $prc"
[ "$rc" = 0 ] && [ "$prc" = 0 ] && echo "OK py311 suite: green, test_pinned_runtime_versions passed"
```
