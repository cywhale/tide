```text
# Release sequence step 2 for v0.3.2: the py311 suite of the merge commit
# ef780ea in a new, separate checkout. The stores are linked as symlinks,
# the VM24 py311 interpreter runs the suite with TIDE_DASK_DISABLE=1, and
# nothing is installed. No root, no pm2, no service, no release directory.
set -u
M=ef780eaca174cb54b790d488c7854d828b56e48e
B=/home/odbadmin/tmp/tide-zarr-6436192.bundle
P=/home/odbadmin/tmp/tide-v032-ef780ea-py311-20261103
O=$P-out
PY=/home/odbadmin/.pyenv/versions/py311/bin/python
T10=/home/odbadmin/python/tide_tpxo10_atlas_v2/data/tpxo10.zarr
T9=/home/odbadmin/python/tide/data/tpxo9.zarr
fail() { echo "FAIL: $*" >&2; exit 1; }
ts() { date '+%F %T %Z'; }

echo "== host $(hostname) user $(id -un) at $(ts) ($(cat /etc/timezone))"
[ ! -e "$P" ] && [ ! -e "$O" ] || fail "$P or $O already exists"
[ -d "$T10" ] && [ -d "$T9" ] || fail "a store is missing"
"$PY" -I -m pytest --version > /dev/null 2>&1 || fail "pytest is not in py311; nothing installed"

echo "== checkout ($(ts))"
git init -q "$P" || fail "git init"
git -C "$P" bundle verify "$B" > /dev/null 2>&1 || fail "bundle verify"
echo "bundle sha256 $(sha256sum "$B" | cut -d' ' -f1)"
git -C "$P" fetch -q "$B" refs/remotes/origin/main:refs/heads/main || fail "fetch"
git -C "$P" checkout -q --detach "$M" || fail "checkout $M"
head=$(git -C "$P" rev-parse HEAD)
echo "HEAD $head"
[ "$head" = "$M" ] || fail "HEAD is $head, not $M"
[ -z "$(git -C "$P" status --porcelain --untracked-files=all)" ] || fail "checkout not clean"
echo "status: clean (no tracked or untracked changes)"

echo "== store links"
mkdir "$P/data" || fail "mkdir data"
ln -s "$T10" "$P/data/tpxo10.zarr" && ln -s "$T9" "$P/data/tpxo9.zarr" || fail "ln"
for l in tpxo10.zarr tpxo9.zarr; do
  echo "$l -> $(readlink "$P/data/$l")"
done
[ "$(readlink "$P/data/tpxo10.zarr")" = "$T10" ] && [ "$(readlink "$P/data/tpxo9.zarr")" = "$T9" ] \
  || fail "a link points elsewhere"
for s in "$T10" "$T9"; do
  echo "store $(stat -c '%U:%G %a %y' "$s") $s: $(find "$s" -type f | wc -l) files"
done
[ -z "$(git -C "$P" status --porcelain --untracked-files=all)" ] || fail "links show up in git status"
echo "status after links: clean (data/ is ignored)"

echo "== interpreter"
"$PY" -I -c 'import sys, zarr, numcodecs, numpy, pytest; print("python", sys.version.split()[0]); print("zarr", zarr.__version__, "numcodecs", numcodecs.__version__, "numpy", numpy.__version__, "pytest", pytest.__version__)'

echo "== suite ($(ts))"
mkdir "$O" || fail "mkdir $O"
touch "$O/start-marker"
cd "$P" || fail "cd"
rc=0
TIDE_DASK_DISABLE=1 "$PY" -m pytest -rs > "$O/pytest.log" 2>&1 || rc=$?
echo "suite exit $rc at $(ts)"
tail -n 40 "$O/pytest.log"
echo "== stores unchanged by the run"
for s in "$T10" "$T9"; do
  n=$(find "$s" -newer "$O/start-marker" | wc -l)
  echo "entries newer than the run start in $s: $n"
done
echo "pytest.log sha256 $(sha256sum "$O/pytest.log" | cut -d' ' -f1) bytes $(wc -c < "$O/pytest.log")"
[ -z "$(git -C "$P" status --porcelain)" ] && echo "checkout still clean (tracked files)" || git -C "$P" status --porcelain | head
echo "== done $(ts) rc=$rc"
exit "$rc"
```
